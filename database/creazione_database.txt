USE forum;
CREATE TABLE ospedali(
	id_ospedale int AUTO_INCREMENT PRIMARY KEY,
    nome varchar(50) NOT null,
    provincia varchar(50) NOT null,
    citta varchar(50) NOT null,
    via varchar(50) NOT null,
    civico int NOT null
);

CREATE TABLE specializzazioni(
	codice int AUTO_INCREMENT PRIMARY KEY,
    nome varchar(50) NOT null,
    descrizione varchar(500)
);

CREATE TABLE dottori(
	id_dottore int AUTO_INCREMENT PRIMARY KEY,
    nome varchar(30) NOT null,
    cognome varchar(30) NOT null,
    eta int(3) NOT null,
    anni_lavoro int(3) NOT null,
    foto_profilo varchar(500),
    specializzazione int NOT null,
    ospedale int NOT null,
    FOREIGN KEY(specializzazione) REFERENCES specializzazioni(codice),
    FOREIGN KEY(ospedale) REFERENCES ospedali(id_ospedale)
);

CREATE TABLE domande(
	id_domanda int AUTO_INCREMENT PRIMARY KEY,
    titolo varchar(50) NOT null,
    corpo varchar(500) NOT null,
    dottore int NOT null,
    FOREIGN KEY(dottore) REFERENCES dottori(id_dottore)
);

CREATE TABLE risposte(
	id_risposta int AUTO_INCREMENT PRIMARY KEY,
    corpo varchar(500) NOT null,
    dottore int NOT null,
    domanda int NOT null,
    FOREIGN KEY(dottore) REFERENCES dottori(id_dottore),
    FOREIGN KEY(domanda) REFERENCES domande(id_domanda)
);
